    // =============================================
    // CALCUL DU BILAN
    // =============================================
    function calculerBilan() {
        const questions = document.querySelectorAll('.bilan-item');
        let total = 0;
        let allAnswered = true;

        questions.forEach((q, index) => {
            const selected = q.querySelector('input[type="radio"]:checked');
            if (selected) {
                total += parseInt(selected.value);
                q.querySelectorAll('.opt').forEach(label => {
                    label.classList.remove('selected');
                });
                selected.closest('.opt').classList.add('selected');
            } else {
                allAnswered = false;
            }
        });

        if (!allAnswered) {
            alert('Veuillez répondre à toutes les affirmations avant de valider.');
            return;
        }

        let badge, level, message, pouvoir;
        if (total >= 26) {
            badge = '🏅';
            level = 'Maître Architecte';
            message = 'Vous maîtrisez parfaitement l\'architecture et le rythme pédagogiques. Vous êtes prêt à construire des formations solides.';
            pouvoir = '🧙 Pouvoir : Vision Structurelle — vous voyez la structure d\'une formation avant même de la construire.';
        } else if (total >= 20) {
            badge = '🏅';
            level = 'Architecte Confirmé';
            message = 'Vous avez de solides bases. Quelques ajustements vous permettront d\'atteindre l\'excellence.';
            pouvoir = '🗡️ Pouvoir : Rythme Parfait — vous sentez instinctivement le bon rythme pour chaque séquence.';
        } else if (total >= 14) {
            badge = '🥉';
            level = 'Apprenti Architecte';
            message = 'Vous avez posé les premières pierres. Revenez sur les points qui vous ont le moins convaincu.';
            pouvoir = '🛡️ Pouvoir : Clairvoyance Pédagogique — vous commencez à identifier les erreurs de séquençage.';
        } else {
            badge = '📖';
            level = 'En chantier';
            message = 'Prenez le temps de revisiter les modules où vous avez le moins confiance avant de vous lancer.';
            pouvoir = '🕯️ Pouvoir : Lumière Naissante — vous commencez à voir clair, la route s\'ouvre devant vous.';
        }

        document.getElementById('resultBadge').textContent = badge;
        document.getElementById('resultLevel').textContent = level;
        document.getElementById('resultScore').textContent = total + ' / 30';
        document.getElementById('resultPouvoir').textContent = pouvoir;
        document.getElementById('resultMessage').textContent = message;
        document.getElementById('resultDetail').textContent = 'Score cumulé Module 2 : ' + total + '/30';

        document.getElementById('resultBox').style.display = 'block';
        document.getElementById('downloadBtn').style.display = 'block';
        document.getElementById('submitBtn').disabled = true;

        window.bilanScore = total;
        window.bilanLevel = level;
        window.bilanMessage = message;
        window.bilanBadge = badge;
        window.bilanPouvoir = pouvoir;

        document.getElementById('resultBox').scrollIntoView({ behavior: 'smooth', block: 'center' });
    }

    function resetBilan() {
        document.querySelectorAll('input[type="radio"]').forEach(input => {
            input.checked = false;
            input.closest('.opt').classList.remove('selected');
        });
        document.getElementById('resultBox').style.display = 'none';
        document.getElementById('downloadBtn').style.display = 'none';
        document.getElementById('submitBtn').disabled = false;
        window.bilanScore = null;
        window.scrollTo({ top: 0, behavior: 'smooth' });
    }

    // =============================================
    // EN-TÊTE DU PDF
    // =============================================
    function addHeader(doc, pageWidth, margin, yPos) {
        const headerY = margin / 2 + 2;

        doc.setFontSize(9);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#080F1E');
        doc.text('UpGrade Learning & Development', margin, headerY);

        const rightText = 'L\'Art de Transmettre #1 - Poser les fondations';
        const textWidth = doc.getTextWidth(rightText);
        doc.setFont('helvetica', 'bold');
        doc.text(rightText, pageWidth - margin - textWidth, headerY);

        doc.setDrawColor('#eef1f6');
        doc.setLineWidth(0.3);
        doc.line(margin, headerY + 4, pageWidth - margin, headerY + 4);

        return headerY + 8;
    }

    // =============================================
    // TÉLÉCHARGEMENT PDF
    // =============================================
    function telechargerPDF() {
        if (typeof window.jspdf === 'undefined') {
            alert('La bibliothèque de génération PDF n\'est pas chargée. Veuillez rafraîchir la page et réessayer.');
            return;
        }

        if (!window.bilanScore && window.bilanScore !== 0) {
            alert('Veuillez d\'abord calculer votre bilan.');
            return;
        }

        const { jsPDF } = window.jspdf;
        const doc = new jsPDF('p', 'mm', 'a4');
        const pageWidth = 210;
        const margin = 20;
        let yPos = margin + 10;

        // En-tête
        yPos = addHeader(doc, pageWidth, margin, yPos) + 6;

        // Bandeau
        doc.setFillColor('#080F1E');
        doc.rect(margin, yPos, pageWidth - (margin * 2), 18, 'F');
        doc.setDrawColor('#D4AF37');
        doc.setLineWidth(0.8);
        doc.line(margin, yPos + 18, pageWidth - margin, yPos + 18);

        doc.setFontSize(9);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#D4AF37');
        doc.text('✦ BILAN DE L\'ARCHITECTE ✦', pageWidth / 2, yPos + 5, { align: 'center' });

        doc.setFontSize(12);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#ffffff');
        doc.text('Auto-Diagnostic · Module 2', pageWidth / 2, yPos + 12.5, { align: 'center' });

        yPos += 24;

        // Sous-titre
        doc.setFontSize(9);
        doc.setFont('helvetica', 'normal');
        doc.setTextColor('#6b7a8f');
        doc.text('L\'Architecture et le Rythme', pageWidth / 2, yPos, { align: 'center' });
        yPos += 6;

        const date = new Date().toLocaleDateString('fr-FR', {
            day: '2-digit',
            month: '2-digit',
            year: 'numeric'
        });
        doc.setFontSize(8);
        doc.setTextColor('#9aabbf');
        doc.text('Date : ' + date, pageWidth / 2, yPos, { align: 'center' });
        yPos += 8;

        // Ligne
        doc.setDrawColor('#eef1f6');
        doc.line(margin, yPos, pageWidth - margin, yPos);
        yPos += 6;

        // Résultat
        doc.setFontSize(10);
        doc.setFont('helvetica', 'normal');
        doc.setTextColor('#6b7a8f');
        doc.text('Votre niveau :', margin, yPos);
        yPos += 5;

        doc.setFontSize(14);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#080F1E');
        doc.text(window.bilanBadge + ' ' + window.bilanLevel, margin, yPos);
        yPos += 6;

        doc.setFontSize(11);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#D4AF37');
        doc.text('Score : ' + window.bilanScore + ' / 30', margin, yPos);
        yPos += 6;

        doc.setFontSize(10);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#080F1E');
        doc.text('Pouvoir :', margin, yPos);
        yPos += 4;
        doc.setFont('helvetica', 'normal');
        doc.setTextColor('#1a2a3a');
        const pouvoirLines = doc.splitTextToSize(window.bilanPouvoir, pageWidth - (margin * 2));
        doc.text(pouvoirLines, margin, yPos);
        yPos += (pouvoirLines.length * 5) + 4;

        doc.setFontSize(10);
        doc.setFont('helvetica', 'normal');
        doc.setTextColor('#1a2a3a');
        const msgLines = doc.splitTextToSize(window.bilanMessage, pageWidth - (margin * 2));
        doc.text(msgLines, margin, yPos);
        yPos += (msgLines.length * 5) + 8;

        // Ligne de séparation
        doc.setDrawColor('#eef1f6');
        doc.line(margin, yPos, pageWidth - margin, yPos);
        yPos += 6;

        // Détail des affirmations
        doc.setFontSize(10);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#080F1E');
        doc.text('Détail de vos réponses :', margin, yPos);
        yPos += 5;

        const questions = [
            '1. Je sais découper mon contenu en séquences, avec une seule idée par séquence.',
            '2. Je comprends pourquoi une séquence doit contenir une seule idée.',
            '3. Je maîtrise l\'alternance théorie/pratique dans mes sessions.',
            '4. J\'ai choisi ma modalité cible et j\'ai adapté mes durées en conséquence.',
            '5. Je sais gérer la fatigue cognitive de mes apprenants.',
            '6. Mon Séquenceur de Cours est une vraie carte de navigation, pas une liste de sujets.'
        ];

        const questionEls = document.querySelectorAll('.bilan-item');

        questionEls.forEach((el, index) => {
            const selected = el.querySelector('input[type="radio"]:checked');
            const value = selected ? selected.value : 'Non répondu';

            if (detailY > 230) {
                doc.addPage();
                detailY = margin + 10;
                detailY = addHeader(doc, pageWidth, margin, detailY) + 6;
            }

            doc.setFontSize(9);

            doc.setTextColor('#080F1E');
            doc.text('Note : ' + value + '/5', margin + 4, detailY);
            detailY += 6;
        });

        // Pied de page
        doc.setFontSize(8);
        doc.setFont('helvetica', 'italic');
        doc.setTextColor('#9aabbf');
        doc.text('Bilan de l\'Architecte · Module 2 · L\'Art de Transmettre', pageWidth / 2, detailY + 10, { align: 'center' });
