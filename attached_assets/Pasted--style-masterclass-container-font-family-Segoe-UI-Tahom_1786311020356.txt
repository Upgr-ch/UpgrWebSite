<script>
    // =============================================
    // EN-TETE DU PDF
    // =============================================
    function addHeader(doc, pageWidth, margin, yPos) {
        const headerY = margin / 2 + 2;

        doc.setFontSize(9);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#080F1E');
        doc.text('UpGrade Learning & Development', margin, headerY);

        const rightText = "L'Art de Transmettre #1 - Poser les fondations";
        const textWidth = doc.getTextWidth(rightText);
        doc.setFont('helvetica', 'bold');
        doc.text(rightText, pageWidth - margin - textWidth, headerY);

        doc.setDrawColor('#eef1f6');
        doc.setLineWidth(0.3);
        doc.line(margin, headerY + 4, pageWidth - margin, headerY + 4);

        return headerY + 8;
    }

    // =============================================
    // TELECHARGEMENT PDF
    // =============================================
    function downloadPDF() {
        if (typeof window.jspdf === 'undefined') {
            alert("La bibliotheque de generation PDF n'est pas chargee. Veuillez rafraichir la page et reessayer.");
            return;
        }

        const { jsPDF } = window.jspdf;
        const doc = new jsPDF('p', 'mm', 'a4');
        const pageWidth = 210;
        const margin = 20;
        let yPos = margin + 10;

        // En-tete
        yPos = addHeader(doc, pageWidth, margin, yPos) + 6;

        // Bandeau
        doc.setFillColor('#080F1E');
        doc.rect(margin, yPos, pageWidth - (margin * 2), 18, 'F');
        doc.setDrawColor('#D4AF37');
        doc.setLineWidth(0.8);
        doc.line(margin, yPos + 18, pageWidth - margin, yPos + 18);

        doc.setFontSize(9);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#D4AF37');
        doc.text('* FICHE SYNTHESE *', pageWidth / 2, yPos + 5, { align: 'center' });

        doc.setFontSize(12);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#ffffff');
        doc.text("Module 02 - L'Architecture et le Rythme", pageWidth / 2, yPos + 12.5, { align: 'center' });

        yPos += 24;

        // Sous-titre
        doc.setFontSize(9);
        doc.setFont('helvetica', 'normal');
        doc.setTextColor('#6b7a8f');
        doc.text('Masterclasse 2h - Module 2', pageWidth / 2, yPos, { align: 'center' });
        yPos += 6;

        const date = new Date().toLocaleDateString('fr-FR', {
            day: '2-digit',
            month: '2-digit',
            year: 'numeric'
        });
        doc.setFontSize(8);
        doc.setTextColor('#9aabbf');
        doc.text('Date : ' + date, pageWidth / 2, yPos, { align: 'center' });
        yPos += 8;

        // Ligne
        doc.setDrawColor('#eef1f6');
        doc.line(margin, yPos, pageWidth - margin, yPos);
        yPos += 6;

        // Introduction
        doc.setFontSize(10);
        doc.setFont('helvetica', 'italic');
        doc.setTextColor('#6b7a8f');
        doc.text("Apres l'ecoute de l'audio du Module 2, consolidez et retenez l'essentiel.", pageWidth / 2, yPos, { align: 'center' });
        yPos += 8;

        // ====== POINTS ======
        const points = [
            { num: '1', title: "Une sequence = une idee", text: "La regle d'or de l'architecture pedagogique : une idee centrale, un exemple concret, une mise en pratique. Pas deux idees. Pas trois. Une. Cette contrainte est un service rendu a votre apprenant." },
            { num: '2', title: "La surcharge cognitive est universelle", text: "Notre cerveau ne peut traiter qu'un nombre limite d'informations nouvelles en parallele. Depasser ce seuil ne signe pas l'intelligence du formateur. Cela signe son manque d'attention a l'apprenant." },
            { num: '3', title: "Le rythme binaire : reception puis action", text: "Alternez systematiquement : apport theorique, puis exercice. Explication, puis application. Ce rythme structure toute pedagogie efficace, quelle que soit la modalite." },
            { num: '4', title: "Le rythme selon la modalite cible", text: "Presentiel : 45-60 min par bloc. Distanciel : 20-25 min, pauses camera coupee. Hybride : cale sur le public le plus contraint, compensations physiques pour le presentiel." },
            { num: '5', title: "Le Sequenceur n'est pas un plan de cours", text: "C'est une carte de navigation. Elle dit dans quel ordre vous allez emmener vos apprenants, quelles escales vous faites, et combien de temps vous passez a chacune." }
        ];

        points.forEach((point) => {
            if (yPos > 230) {
                doc.addPage();
                yPos = margin + 10;
                yPos = addHeader(doc, pageWidth, margin, yPos) + 6;
            }

            // Numero en cercle
            doc.setFillColor('#080F1E');
            doc.circle(margin + 8, yPos - 2, 5, 'F');
            doc.setFontSize(9);
            doc.setFont('helvetica', 'bold');
            doc.setTextColor('#ffffff');
            doc.text(point.num, margin + 8, yPos + 2, { align: 'center' });

            // Titre
            doc.setFontSize(10);
            doc.setFont('helvetica', 'bold');
            doc.setTextColor('#080F1E');
            doc.text(point.title, margin + 16, yPos);
            yPos += 5;

            // Texte
            doc.setFontSize(9);
            doc.setFont('helvetica', 'normal');
            doc.setTextColor('#1a2a3a');
            const textLines = doc.splitTextToSize(point.text, pageWidth - (margin * 2));
            doc.text(textLines, margin, yPos);
            yPos += (textLines.length * 4.5) + 6;
        });

        // ====== QUESTIONS ======
        if (yPos > 210) {
            doc.addPage();
            yPos = margin + 10;
            yPos = addHeader(doc, pageWidth, margin, yPos) + 6;
        }

        // Titre questions avec encadre
        doc.setFillColor('#080F1E');
        doc.roundedRect(margin + 30, yPos - 3, 90, 7, 2, 2, 'F');
        doc.setFontSize(9);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#ffffff');
        doc.text('Questions de reflexion personnelle', margin + 75, yPos + 1, { align: 'center' });
        yPos += 10;

        const questions = [
            "1. Dans votre contenu actuel, y a-t-il des sequences qui contiennent trop d'idees ? Lesquelles ?",
            "2. Avez-vous tendance a sous-estimer ou surestimer ce que vos apprenants peuvent absorber en une heure ?",
            "3. Le rythme que vous envisagez est-il vraiment adapte aux contraintes de votre modalite cible ?"
        ];

        questions.forEach((q) => {
            doc.setFontSize(9);
            doc.setFont('helvetica', 'normal');
            doc.setTextColor('#1a2a3a');
            doc.setTextColor('#D4AF37');
            doc.text('*', margin, yPos);
            doc.setTextColor('#1a2a3a');
            const qLines = doc.splitTextToSize(q, pageWidth - (margin * 2) - 6);
            doc.text(qLines, margin + 6, yPos);
            yPos += (qLines.length * 4.5) + 4;
        });

        // Pied de page
        doc.setFontSize(8);
        doc.setFont('helvetica', 'italic');
        doc.setTextColor('#9aabbf');
        doc.text("Fiche Synthese - Module 2 · L'Art de Transmettre", pageWidth / 2, yPos + 10, { align: 'center' });

        doc.save('Fiche_Synthese_M2.pdf');
    }
</script>
