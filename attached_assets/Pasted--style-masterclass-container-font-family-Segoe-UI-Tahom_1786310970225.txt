<script>
    // =============================================
    // ÉTAT
    // =============================================
    let sequenceCount = 0;
    let modaliteActuelle = 'presentiel';

    const formats = ['Théorie', 'Pratique', 'Atelier', 'Quiz', 'Synthèse', 'Mise en situation', 'Débrief', 'Exercice'];

    const dureesRecommandees = {
        presentiel: { min: 45, max: 60, label: '45-60 min' },
        distanciel: { min: 20, max: 25, label: '20-25 min' },
        hybride: { min: 20, max: 25, label: '20-25 min' }
    };

    // =============================================
    // INITIALISATION
    // =============================================
    document.addEventListener('DOMContentLoaded', function() {
        // Ajouter 3 lignes par défaut
        for (let i = 0; i < 3; i++) {
            ajouterLigne();
        }
        // Mettre à jour les durées
        mettreAJourDurees();
    });

    // =============================================
    // GESTION DES LIGNES
    // =============================================
    function ajouterLigne() {
        const tbody = document.getElementById('sequenceBody');
        const tr = document.createElement('tr');
        tr.id = 'row_' + sequenceCount;

        const modalite = document.querySelector('input[name="modalite_cible"]:checked').value;
        const dureeDefaut = dureesRecommandees[modalite].min;

        tr.innerHTML = `
            <td><input type="text" class="seq-titre" placeholder="Ex: Prioriser ses tâches" data-row="${sequenceCount}"></td>
            <td><input type="text" class="seq-objectif" placeholder="Ex: Classer ses tâches dans la matrice" data-row="${sequenceCount}"></td>
            <td>
                <select class="seq-format" data-row="${sequenceCount}">
                    ${formats.map(f => `<option value="${f}">${f}</option>`).join('')}
                </select>
            </td>
            <td><input type="number" class="seq-duree" value="${dureeDefaut}" min="5" max="90" data-row="${sequenceCount}" style="width:60px;"> min</td>
            <td>
                <select class="seq-modalite" data-row="${sequenceCount}">
                    <option value="presentiel">Présentiel</option>
                    <option value="distanciel" ${modalite === 'distanciel' ? 'selected' : ''}>Distanciel</option>
                    <option value="hybride" ${modalite === 'hybride' ? 'selected' : ''}>Hybride</option>
                </select>
            </td>
            <td><button class="btn-remove-row" onclick="supprimerLigne(${sequenceCount})" ${sequenceCount < 3 ? 'disabled style="opacity:0.3;"' : ''}>✕</button></td>
        `;

        tbody.appendChild(tr);
        sequenceCount++;
        mettreAJourDurees();
    }

    function supprimerLigne(rowId) {
        const row = document.getElementById('row_' + rowId);
        if (row && document.querySelectorAll('#sequenceBody tr').length > 1) {
            row.remove();
            mettreAJourDurees();
        }
    }

    // =============================================
    // MISE À JOUR DES DURÉES
    // =============================================
    function mettreAJourDurees() {
        const modalite = document.querySelector('input[name="modalite_cible"]:checked').value;
        modaliteActuelle = modalite;
        const dureeDefaut = dureesRecommandees[modalite].min;

        document.querySelectorAll('.seq-duree').forEach(input => {
            if (!input.dataset.userValue) {
                input.value = dureeDefaut;
            }
        });

        document.querySelectorAll('.seq-modalite').forEach(select => {
            select.value = modalite;
        });
    }

    // =============================================
    // VALIDATION DU SÉQUENCEUR
    // =============================================
    function validerSequenceur() {
        const rows = document.querySelectorAll('#sequenceBody tr');
        if (rows.length === 0) {
            alert('Ajoutez au moins une séquence avant de valider.');
            return;
        }

        let resultats = {
            regle1: { ok: true, details: [] },
            regle2: { ok: true, details: [] },
            regle3: { ok: true, details: [] }
        };

        let titres = [];
        let formatsList = [];

        rows.forEach((row, index) => {
            const titre = row.querySelector('.seq-titre').value.trim();
            const objectif = row.querySelector('.seq-objectif').value.trim();
            const format = row.querySelector('.seq-format').value;
            const duree = parseInt(row.querySelector('.seq-duree').value);
            const modalite = row.querySelector('.seq-modalite').value;

            // Nettoyer les champs
            row.querySelector('.seq-titre').classList.remove('error', 'success');
            row.querySelector('.seq-objectif').classList.remove('error', 'success');
            row.querySelector('.seq-duree').classList.remove('error', 'success');

            // Règle 1 : Une idée par séquence
            if (titre === '') {
                resultats.regle1.ok = false;
                resultats.regle1.details.push('Séquence ' + (index + 1) + ' : titre manquant');
                row.querySelector('.seq-titre').classList.add('error');
            } else if (titre.toLowerCase().includes(' et ') || titre.includes(' & ') || titre.includes('+')) {
                resultats.regle1.ok = false;
                resultats.regle1.details.push('Séquence ' + (index + 1) + ' : "' + titre + '" semble contenir plusieurs idées');
                row.querySelector('.seq-titre').classList.add('error');
            } else {
                row.querySelector('.seq-titre').classList.add('success');
            }

            if (objectif === '') {
                resultats.regle1.ok = false;
                resultats.regle1.details.push('Séquence ' + (index + 1) + ' : objectif manquant');
                row.querySelector('.seq-objectif').classList.add('error');
            } else {
                row.querySelector('.seq-objectif').classList.add('success');
            }

            // Règle 3 : Durées adaptées
            const dureeMin = dureesRecommandees[modalite].min;
            const dureeMax = dureesRecommandees[modalite].max;
            if (duree < dureeMin || duree > dureeMax) {
                resultats.regle3.ok = false;
                resultats.regle3.details.push('Séquence ' + (index + 1) + ' : durée ' + duree + ' min (recommandé : ' + dureeMin + '-' + dureeMax + ' min pour ' + modalite + ')');
                row.querySelector('.seq-duree').classList.add('error');
            } else {
                row.querySelector('.seq-duree').classList.add('success');
            }

            titres.push(titre);
            formatsList.push(format);
        });

        // Règle 2 : Alternance théorie/pratique
        let theorieConsecutive = 0;
        for (let i = 0; i < formatsList.length; i++) {
            if (formatsList[i] === 'Théorie') {
                theorieConsecutive++;
                if (theorieConsecutive > 2) {
                    resultats.regle2.ok = false;
                    resultats.regle2.details.push('3 théories d\'affilée (séquences ' + (i - 1) + ', ' + i + ', ' + (i + 1) + '). Ajoutez une pratique.');
                    break;
                }
            } else {
                theorieConsecutive = 0;
            }
        }

        // Affichage des résultats
        const validationDiv = document.getElementById('validationResult');
        const detailsDiv = document.getElementById('validationDetails');
        const scoreDiv = document.getElementById('validationScore');

        let html = '';
        let score = 0;
        const regles = [
            { id: 'regle1', label: 'Une idée par séquence', ok: resultats.regle1.ok },
            { id: 'regle2', label: 'Alternance théorie/pratique', ok: resultats.regle2.ok },
            { id: 'regle3', label: 'Durées adaptées', ok: resultats.regle3.ok }
        ];

        regles.forEach(regle => {
            const details = resultats[regle.id].details;
            const status = regle.ok ? '✅' : '❌';
            const statusClass = regle.ok ? 'ok' : 'ko';
            html += `
                <div class="vr-ligne">
                    <span class="vr-status ${statusClass}">${status} ${regle.label}</span>
                    ${details.length > 0 ? '<span style="font-size:13px;color:#6b7a8f;">' + details.join(' · ') + '</span>' : '<span style="font-size:13px;color:#2d8f5c;">OK</span>'}
                </div>
            `;
            if (regle.ok) score++;
        });

        detailsDiv.innerHTML = html;

        let niveau = '';
        if (score === 3) niveau = 'Architecte Confirmé · Votre Séquenceur est prêt à être déployé.';
        else if (score === 2) niveau = 'Plan à affiner · Corrigez les points signalés ci-dessus.';
        else niveau = 'Chantier à reprendre · Revoyez votre structure en profondeur.';

        scoreDiv.textContent = score + ' / 3 · ' + niveau;

        validationDiv.classList.add('open');
        document.getElementById('downloadBtn').style.display = 'block';
        validationDiv.scrollIntoView({ behavior: 'smooth', block: 'center' });
    }

    function addHeader(doc, pageWidth, margin, yPos) {
        const headerY = margin / 2 + 2;
        doc.setFontSize(9);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#080F1E');
        doc.text('UpGrade Learning & Development', margin, headerY);

        const rightText = 'L\'Art de Transmettre #1 - Poser les fondations';
        const textWidth = doc.getTextWidth(rightText);
        doc.setFont('helvetica', 'bold');
        doc.text(rightText, pageWidth - margin - textWidth, headerY);

        doc.setDrawColor('#eef1f6');
        doc.setLineWidth(0.3);
        doc.line(margin, headerY + 4, pageWidth - margin, headerY + 4);
        return headerY + 8;
    }

    function telechargerPDF() {
        if (typeof window.jspdf === 'undefined') {
            alert('La bibliothèque de génération PDF n\'est pas chargée. Veuillez rafraîchir la page et réessayer.');
            return;
        }

        const rows = document.querySelectorAll('#sequenceBody tr');
        if (rows.length === 0) {
            alert('Ajoutez au moins une séquence avant de télécharger.');
            return;
        }

        const { jsPDF } = window.jspdf;
        const doc = new jsPDF('p', 'mm', 'a4');
        const pageWidth = 210;
        const margin = 20;
        let yPos = margin + 10;

        yPos = addHeader(doc, pageWidth, margin, yPos) + 6;

        doc.setFillColor('#080F1E');
        doc.rect(margin, yPos, pageWidth - (margin * 2), 18, 'F');
        doc.setDrawColor('#D4AF37');
        doc.setLineWidth(0.8);
        doc.line(margin, yPos + 18, pageWidth - margin, yPos + 18);

        doc.setFontSize(9);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#D4AF37');
        doc.text('EXERCICE 4 · SÉQUENCEUR DE COURS', pageWidth / 2, yPos + 5, { align: 'center' });

        doc.setFontSize(12);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#ffffff');
        doc.text('Mon séquenceur', pageWidth / 2, yPos + 12.5, { align: 'center' });

        yPos += 24;

        const date = new Date().toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit', year: 'numeric' });
        doc.setFontSize(8);
        doc.setTextColor('#9aabbf');
        doc.text('Date : ' + date, pageWidth / 2, yPos, { align: 'center' });
        yPos += 8;

        const modalite = document.querySelector('input[name="modalite_cible"]:checked').value;
        const modaliteLabels = { presentiel: 'Présentiel', distanciel: 'Distanciel', hybride: 'Hybride' };
        doc.setFontSize(9);
        doc.setTextColor('#6b7a8f');
        doc.text('Modalité cible : ' + modaliteLabels[modalite], pageWidth / 2, yPos, { align: 'center' });
        yPos += 8;

        doc.setDrawColor('#eef1f6');
        doc.line(margin, yPos, pageWidth - margin, yPos);
        yPos += 8;

        // Tableau
        const colWidths = [38, 45, 28, 20, 25];
        const headers = ['Titre', 'Objectif observable', 'Format', 'Durée', 'Modalité'];

        doc.setFontSize(9);
        doc.setFont('helvetica', 'bold');
        doc.setTextColor('#ffffff');
        doc.setFillColor('#080F1E');
        doc.rect(margin, yPos, pageWidth - (margin * 2), 7, 'F');

        let xPos = margin;
        headers.forEach((h, i) => {
            doc.text(h, xPos + 2, yPos + 4.5);
            xPos += colWidths[i];
        });
        yPos += 7;

        doc.setFont('helvetica', 'normal');
        doc.setTextColor('#1a2a3a');

        rows.forEach((row, index) => {
            if (yPos > 240) {
                doc.addPage();
                yPos = margin + 10;
                yPos = addHeader(doc, pageWidth, margin, yPos) + 6;
            }

            const titre = row.querySelector('.seq-titre').value || 'Non renseigné';
            const objectif = row.querySelector('.seq-objectif').value || 'Non renseigné';
            const format = row.querySelector('.seq-format').value;
            const duree = row.querySelector('.seq-duree').value || '?';
            const modaliteRow = row.querySelector('.seq-modalite').value;
            const modaliteRowLabel = modaliteLabels[modaliteRow] || modaliteRow;

            const cells = [titre, objectif, format, duree + ' min', modaliteRowLabel];
            let xPosRow = margin;

            // Fond alterné
            if (index % 2 === 0) {
                doc.setFillColor('#f7f8fc');
                doc.rect(margin, yPos - 1, pageWidth - (margin * 2), 6.5, 'F');
            }

            cells.forEach((c, i) => {
                const text = c.length > 25 ? c.substring(0, 22) + '...' : c;
                doc.text(text, xPosRow + 2, yPos + 4);
                xPosRow += colWidths[i];
            });
            yPos += 7;
        });

        yPos += 4;

        // Pied de page
        doc.setFontSize(8);
        doc.setFont('helvetica', 'italic');
        doc.setTextColor('#9aabbf');
        doc.text('Séquenceur de Cours · L\'Art de Transmettre #1', pageWidth / 2, yPos + 10, { align: 'center' });

        doc.save('Sequenceur_de_Cours.pdf');
    }
